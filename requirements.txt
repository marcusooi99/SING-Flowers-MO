# Minimal dependencies for running the LeafMachine2 Plant Component Detector
# (YOLOv5x6) standalone via torch.hub, without LeafMachine2's own heavy stack
# (no Detectron2, no vit-pytorch/DocEnTr, no pycocotools).

# torch is capped at <2.6.0: PyTorch 2.6 changed torch.load's default from
# weights_only=False to True, which breaks loading LeafMachine2's weights
# (saved as a full pickled model object, yolov5 v7.0-era style, not a plain
# state dict). Capping the version avoids needing to patch around it.
torch>=2.0.0,<2.6.0
torchvision>=0.15.0,<0.21.0
ultralytics>=8.0.0
opencv-python>=4.7.0
pillow>=9.0.0
numpy>=1.23.0
pandas>=1.5.0
pyyaml>=6.0
matplotlib>=3.6.0
seaborn>=0.12.0
scipy>=1.9.0
tqdm>=4.64.0
requests>=2.28.0
gitpython>=3.1.0
ipython>=8.0.0
psutil>=5.9.0
thop>=0.1.1
setuptools<81
